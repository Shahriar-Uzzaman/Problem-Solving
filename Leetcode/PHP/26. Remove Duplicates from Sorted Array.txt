if(count($nums) === 0)
        {
            return 0;
        }

        $index = 1;

        for ($readIndex = 1; $readIndex<count($nums); $readIndex++)
        {
            if ($nums[$readIndex] !== $nums[$readIndex-1])
            {
                $nums[$index] = $nums[$readIndex];
                $index++;
            }
        }
        return $index;
